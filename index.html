import { test, expect } from '@playwright/test';

test.describe('Frontend Robustness Guard', () => {
  const consoleErrors = [];

  // 1. Setup Global Console Listener
  test.beforeEach(async ({ page }) => {
    page.on('console', msg => {

      if (msg.type() === 'error') {
        consoleErrors.push(msg.text());
      }
    });
  });

  // 2. Fail Test On Any "Silent" Errors
  test.afterEach(() => {

    if (consoleErrors.length > 0) {
      throw new Error(`Test Failed! Caught ${consoleErrors.length} console errors.`);
    }
  });
});
