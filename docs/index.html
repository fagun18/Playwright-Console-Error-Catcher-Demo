test.describe('Frontend Guard', () => {
  const errors = [];

  // 1. Listen for console errors
  test.beforeEach(async ({ page }) => {
    page.on('console', msg => {

      if (msg.type() === 'error') {
        errors.push(msg.text());
      }
    });
  });

  // 2. Fail test if errors found
  test.afterEach(() => {

    if (errors.length > 0) {
      throw new Error(`Caught ${errors.length} errors!`);
    }
  });
});
